# HELPER FUNCTIONS
# ============================================================

def normalize_text(text):
    """Convert text to lowercase and remove extra spaces."""
    return " ".join(text.lower().split())


def title_matches(job_title):
    """
    Check whether the job title matches one of the
    target finance-related job titles.
    """

    title = normalize_text(job_title)

    for target in TARGET_JOB_TITLES:
        target_normalized = normalize_text(target)

        if target_normalized in title:
            return True

    return False


def location_matches(location_text):
    """
    Check whether the job location matches one of
    the target Indian locations.
    """

    location = normalize_text(location_text)

    for target in TARGET_LOCATIONS:
        target_normalized = normalize_text(target)

        if target_normalized in location:
            return True

    return False


def extract_experience(text):
    """
    Try to extract experience range from job text.

    Examples:
        0 - 2 years
        1 - 3 years
        2 years
        0-1 yrs
    """

    text = normalize_text(text)

    patterns = [
        r"(\d+)\s*-\s*(\d+)\s*years?",
        r"(\d+)\s*-\s*(\d+)\s*yrs?",
        r"(\d+)\s*years?",
        r"(\d+)\s*yrs?"
    ]

    for pattern in patterns:
        match = re.search(pattern, text)

        if match:
            numbers = match.groups()

            if len(numbers) == 2:
                min_exp = int(numbers[0])
                max_exp = int(numbers[1])

                return min_exp, max_exp

            if len(numbers) == 1:
                exp = int(numbers[0])

                return exp, exp

    return None


def experience_matches(job_text):
    """
    Check whether the job experience overlaps
    with our desired 0-2 years range.
    """

    experience = extract_experience(job_text)

    # If experience isn't visible on the card,
    # don't automatically reject the job.
    if experience is None:
        return True

    job_min, job_max = experience

    # Check whether the job's experience range
    # overlaps our target range.
    if job_max >= MIN_EXPERIENCE and job_min <= MAX_EXPERIENCE:
        return True

    return False
